cmake_minimum_required(VERSION 3.20)
project(FrameStride VERSION 1.0.2 LANGUAGES CXX)

include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)

option(FRAMESTRIDE_NATIVE "Build SteamVR and OpenXR integration on Linux" ${UNIX})
set(FRAMESTRIDE_DEPS "${CMAKE_BINARY_DIR}/deps" CACHE PATH "Directory for verified dependency downloads")
include(cmake/dependencies.cmake)

add_library(stride_core STATIC src/core.cpp src/config.cpp)
target_include_directories(stride_core PUBLIC src)
target_include_directories(stride_core SYSTEM PRIVATE "${FRAMESTRIDE_DEPS}")
target_compile_definitions(stride_core PUBLIC FRAMESTRIDE_VERSION="${PROJECT_VERSION}")

if(FRAMESTRIDE_NATIVE)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The native application needs Linux. Use cmake/zig-linux.cmake to cross-compile.")
    endif()
    set(THREADS_PREFER_PTHREAD_FLAG ON)
    find_package(Threads REQUIRED)
    add_library(stride_ipc STATIC src/ipc.cpp)
    target_include_directories(stride_ipc PUBLIC src)

    add_library(stride_dashboard STATIC src/dashboard.cpp)
    target_include_directories(stride_dashboard SYSTEM PRIVATE "${FRAMESTRIDE_DEPS}")
    target_link_libraries(stride_dashboard PUBLIC stride_core stride_ipc)

    add_executable(framestride src/main.cpp src/legacy_apps.cpp)
    target_include_directories(framestride SYSTEM PRIVATE "${FRAMESTRIDE_DEPS}")
    target_link_libraries(framestride PRIVATE stride_dashboard Threads::Threads ${CMAKE_DL_LIBS})

    add_library(driver_framestride SHARED src/driver.cpp)
    set_target_properties(driver_framestride PROPERTIES PREFIX "")
    target_include_directories(driver_framestride SYSTEM PRIVATE "${FRAMESTRIDE_DEPS}")
    target_link_libraries(driver_framestride PRIVATE stride_ipc)

    add_library(framestride_openxr SHARED src/openxr_layer.cpp)
    target_include_directories(framestride_openxr SYSTEM PRIVATE "${FRAMESTRIDE_DEPS}")
    target_link_libraries(framestride_openxr PRIVATE stride_core stride_ipc Threads::Threads)
    target_link_options(framestride PRIVATE -pie)
    # Zig's statically linked libc++ can register process-lifetime callbacks from a DSO.
    # Close devices and sessions immediately, and retain their code until host process exit.
    target_link_options(driver_framestride PRIVATE -Wl,-z,nodelete)
    target_link_options(framestride_openxr PRIVATE -Wl,-z,nodelete)
    foreach(binary framestride driver_framestride framestride_openxr)
        target_link_options(${binary} PRIVATE "$<$<CONFIG:Release>:-Wl,--strip-debug>")
    endforeach()

    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
        set(driver_platform linuxarm64)
    elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
        set(driver_platform linux64)
    else()
        message(FATAL_ERROR "No OpenVR driver layout for ${CMAKE_SYSTEM_PROCESSOR}")
    endif()
    install(TARGETS framestride RUNTIME DESTINATION payload)
    install(TARGETS driver_framestride LIBRARY DESTINATION "payload/driver/framestride/bin/${driver_platform}")
    install(TARGETS framestride_openxr LIBRARY DESTINATION payload)
    install(DIRECTORY packaging/driver/ DESTINATION payload/driver/framestride)
    install(DIRECTORY packaging/probe packaging/assets DESTINATION payload)
    install(FILES "${FRAMESTRIDE_DEPS}/InterVariable.ttf" DESTINATION payload/assets)
    install(FILES packaging/framestride.vrmanifest packaging/default-settings.json DESTINATION payload)
    install(PROGRAMS scripts/treadmill.sh DESTINATION payload)
    install(FILES README.md LICENSE DESTINATION .)
    install(FILES docs/INPUT.md DESTINATION docs)
    install(FILES "${FRAMESTRIDE_DEPS}/openvr-LICENSE" "${FRAMESTRIDE_DEPS}/json-LICENSE"
            "${FRAMESTRIDE_DEPS}/openxr-LICENSE" "${FRAMESTRIDE_DEPS}/Inter-LICENSE" packaging/stb-LICENSE
            DESTINATION licenses)
    install(PROGRAMS scripts/install.sh scripts/uninstall.sh scripts/update.sh scripts/test-on-frame.sh DESTINATION .)
    install(FILES scripts/install.ps1 DESTINATION .)
endif()

if(BUILD_TESTING)
    add_executable(stride_tests tests/core_tests.cpp src/ipc.cpp)
    target_link_libraries(stride_tests PRIVATE stride_core)
    add_test(NAME motion_config_safety COMMAND stride_tests)
    if(FRAMESTRIDE_NATIVE)
        add_test(NAME application_version COMMAND framestride --version)
        add_executable(stride_xr_tests tests/openxr_tests.cpp)
        target_include_directories(stride_xr_tests SYSTEM PRIVATE "${FRAMESTRIDE_DEPS}")
        target_link_libraries(stride_xr_tests PRIVATE stride_ipc Threads::Threads ${CMAKE_DL_LIBS})
        add_test(NAME openxr_action_delivery COMMAND stride_xr_tests $<TARGET_FILE:framestride_openxr>)
        add_executable(stride_driver_tests tests/driver_tests.cpp)
        target_include_directories(stride_driver_tests SYSTEM PRIVATE "${FRAMESTRIDE_DEPS}")
        target_link_libraries(stride_driver_tests PRIVATE stride_ipc Threads::Threads ${CMAKE_DL_LIBS})
        add_test(NAME steamvr_driver_submission COMMAND stride_driver_tests $<TARGET_FILE:driver_framestride>)
        add_executable(stride_dashboard_tests tests/dashboard_tests.cpp)
        target_link_libraries(stride_dashboard_tests PRIVATE stride_dashboard)
        add_test(NAME dashboard_controls COMMAND stride_dashboard_tests "${FRAMESTRIDE_DEPS}/InterVariable.ttf")
    endif()
endif()
